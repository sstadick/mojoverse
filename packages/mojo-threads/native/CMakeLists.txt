cmake_minimum_required(VERSION 3.20)
# Use the Xcode SDK toolchain on macOS, not Conda's incompatible LTO driver.
if(CMAKE_HOST_APPLE)
  set(CMAKE_C_COMPILER /usr/bin/clang CACHE FILEPATH "Apple Clang" FORCE)
endif()
project(mojo_threads_native LANGUAGES C)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use Clang for native builds")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
add_library(mojo_threads_native STATIC threads.c)
target_compile_options(mojo_threads_native PRIVATE -O3 -march=native -Wall -Wextra -Werror)
target_link_libraries(mojo_threads_native PRIVATE Threads::Threads)
install(TARGETS mojo_threads_native ARCHIVE DESTINATION lib)
install(FILES ../LICENSE DESTINATION share/licenses/mojo-threads-native)
