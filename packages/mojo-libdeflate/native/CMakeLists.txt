cmake_minimum_required(VERSION 3.20)
# Use the Xcode SDK toolchain on macOS, not Conda's incompatible LTO driver.
if(CMAKE_HOST_APPLE)
  set(CMAKE_C_COMPILER /usr/bin/clang CACHE FILEPATH "Apple Clang" FORCE)
endif()
project(mojo_libdeflate_native LANGUAGES C)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use Clang for matched native codec builds")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-O3 -march=native)
set(LIBDEFLATE_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_GZIP OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/libdeflate)
install(TARGETS libdeflate_static ARCHIVE DESTINATION lib)
install(FILES vendor/libdeflate/libdeflate.h DESTINATION include)
install(FILES vendor/libdeflate/COPYING DESTINATION share/licenses/mojo-libdeflate-native)
